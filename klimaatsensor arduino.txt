#include <SoftwareSerial.h>
#include <SimpleDHT.h>
#include <Wire.h>
#include "Adafruit_SGP30.h"

#define DHT_PIN     3
#define BT_RX_PIN   10
#define BT_TX_PIN   11

SimpleDHT11 dht11(DHT_PIN);
Adafruit_SGP30 sgp;
SoftwareSerial bluetooth(BT_RX_PIN, BT_TX_PIN);

unsigned long vorigeMillis = 0;
const unsigned long interval = 2000;

uint32_t berekenAbsoluteVochtigheid(float temperatuur, float relatieveVochtigheid) {
  float absVochtigheid = (relatieveVochtigheid / 100.0) * 6.112 *
    exp((17.67 * temperatuur) / (temperatuur + 243.5)) *
    2.1674 / (273.15 + temperatuur);
  uint16_t ah_scaled = (uint16_t)(absVochtigheid * 256.0);
  return (uint32_t)ah_scaled << 8 | 0x00;
}

void setup() {
  Serial.begin(9600);
  bluetooth.begin(9600);

  Serial.println("Klimaatsensor opstarten...");
  bluetooth.println("Klimaatsensor verbonden!");

  if (!sgp.begin()) {
    Serial.println("FOUT: SGP30 niet gevonden!");
    bluetooth.println("FOUT: SGP30 niet gevonden!");
    while (true);
  }
  Serial.println("SGP30 gevonden.");
  delay(2000);
}

void loop() {
  unsigned long huidigeMillis = millis();

  if (huidigeMillis - vorigeMillis >= interval) {
    vorigeMillis = huidigeMillis;

    bluetooth.end();
    byte temp = 0;
    byte vocht = 0;
    int err = dht11.read(&temp, &vocht, NULL);
    bluetooth.begin(9600);

    if (err != SimpleDHTErrSuccess) {
      Serial.print("FOUT DHT11: ");
      Serial.println(err);
      bluetooth.print("FOUT DHT11: ");
      bluetooth.println(err);
    }

    if (!sgp.IAQmeasure()) {
      Serial.println("FOUT: SGP30 meting mislukt!");
      bluetooth.println("FOUT: SGP30 meting mislukt!");
      return;
    }

    uint16_t eco2 = sgp.eCO2;
    uint16_t tvoc = sgp.TVOC;

    if (err == SimpleDHTErrSuccess) {
      sgp.setHumidity(berekenAbsoluteVochtigheid((float)temp, (float)vocht));
    }

    String data = "";
    if (err == SimpleDHTErrSuccess) {
      data += "T:" + String((int)temp) + ".0C ";
      data += "RH:" + String((int)vocht) + ".0% ";
    }
    data += "CO2:" + String(eco2) + "ppm ";
    data += "TVOC:" + String(tvoc) + "ppb";

    Serial.println(data);
    bluetooth.println(data);

    String status = "";
    if (eco2 < 1000 && tvoc < 220) {
      status = "Luchtkwaliteit: GOED";
    } else if (eco2 < 2000 && tvoc < 660) {
      status = "Luchtkwaliteit: MATIG";
    } else {
      status = "Luchtkwaliteit: SLECHT - ventileer!";
    }
    Serial.println(status);
    bluetooth.println(status);

    static uint16_t teller = 0;
    teller++;
    if (teller % 15 == 0) {
      uint16_t TVOC_base, eCO2_base;
      if (sgp.getIAQBaseline(&eCO2_base, &TVOC_base)) {
        Serial.print("Basislijn eCO2: 0x"); Serial.print(eCO2_base, HEX);
        Serial.print(" TVOC: 0x"); Serial.println(TVOC_base, HEX);
      }
    }
  }
}
